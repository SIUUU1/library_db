-- [Q03] 반납 예정일이 임박한 미반납 대여
-- 설명: 아직 반납되지 않은(returned_at IS NULL) 대여 건을 반납 예정일이 빠른 순으로 확인한다.
SELECT rental_id, member_id, book_id, rented_at, due_date, status
FROM rental
WHERE returned_at IS NULL
ORDER BY due_date ASC;
