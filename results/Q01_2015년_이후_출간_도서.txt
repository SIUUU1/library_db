-- [Q01] 2015년 이후 출간 도서
-- 설명: WHERE 조건으로 2015년 이후 출간된 도서만 골라 최신 출간순으로 확인한다.
SELECT book_id, title, author, published_year
FROM book
WHERE published_year >= 2015
ORDER BY published_year DESC, title;
