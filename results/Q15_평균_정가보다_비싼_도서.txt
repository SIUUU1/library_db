-- [Q15] 평균 정가보다 비싼 도서
-- 설명: 스칼라 서브쿼리로 전체 평균 정가를 구하고, 그보다 비싼 도서만 확인한다.
SELECT title,
       price,
       (SELECT ROUND(AVG(price), 0) FROM book) AS overall_avg_price
FROM book
WHERE price > (SELECT AVG(price) FROM book)
ORDER BY price DESC;
