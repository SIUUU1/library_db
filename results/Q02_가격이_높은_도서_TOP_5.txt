-- [Q02] 가격이 높은 도서 TOP 5
-- 설명: ORDER BY + LIMIT로 정가가 가장 비싼 도서 5권을 확인한다.
SELECT title, author, price
FROM book
ORDER BY price DESC
LIMIT 5;
