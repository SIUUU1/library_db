-- [Q09] (LEFT JOIN) 한 번도 대여되지 않은 도서 포함 전체 도서의 대여 이력
-- 설명: book을 기준으로 LEFT JOIN 해, 대여 기록이 없는 도서도 rental 쪽이 NULL로 나오는 것을 확인한다.
SELECT b.book_id,
       b.title,
       r.rental_id,
       r.rented_at
FROM book b
LEFT JOIN rental r ON b.book_id = r.book_id
ORDER BY b.book_id, r.rented_at;
