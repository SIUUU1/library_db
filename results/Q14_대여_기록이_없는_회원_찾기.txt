-- [Q14] 대여 기록이 없는 회원 찾기
-- 설명: NOT IN 서브쿼리로 rental 테이블에 한 번도 등장하지 않은 회원을 확인한다.
SELECT member_id, name, email, joined_at
FROM member
WHERE member_id NOT IN (SELECT member_id FROM rental)
ORDER BY member_id;
