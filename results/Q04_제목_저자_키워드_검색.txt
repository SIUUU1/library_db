-- [Q04] 제목·저자 키워드 검색
-- 설명: LIKE로 제목에 'SQL' 또는 '코드'가 들어가거나 저자가 '한강'인 도서를 검색한다.
SELECT book_id, title, author
FROM book
WHERE title LIKE '%SQL%'
   OR title LIKE '%코드%'
   OR author = '한강'
ORDER BY book_id;
